# nhận diện IOS/IOS-XE. KHÔNG dùng cú pháp NX-OS (vpc, vrf all,
# system resources...) — xem cisco_nxos_commands.txt cho Nexus.

# --- Hệ thống & tài nguyên ---
show version
show inventory
show processes cpu sorted
show processes memory sorted
show env all
dir flash:

# --- Lớp 3 ---
show ip interface brief
show ip route
# show vrf (Layer 3 / VRF-enabled devices only)

# --- Lớp 2 / cổng ---
show interfaces status
show interfaces trunk
show vlan brief
show etherchannel summary
show mac address-table

# --- STP & láng giềng ---
show spanning-tree summary
show cdp neighbors
show lldp neighbors

# --- Bảo mật, stack & nhật ký ---
show port-security
show switch
# show logging (dumps entire buffer; use 'netops exec' when full event history is needed)
